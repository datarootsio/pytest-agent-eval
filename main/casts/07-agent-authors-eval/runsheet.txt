import pytest

from pytest_agent_eval import AgentReply, ToolCall

BOOKED = AgentReply(
    "Booking confirmed for 10am. Reference BK-1234.",
    [ToolCall("create_booking", {"time": "10am"})],
)
RESCHEDULED = AgentReply(
    "Rescheduled to 11am. Booking BK-1234 is still confirmed.",
    [ToolCall("update_booking", {"time": "11am"})],
)


@pytest.fixture
def llm_eval_agent():
    # Stub booking agent: fixed replies, so the evals need no API key.
    async def agent(history):
        return RESCHEDULED if "11am" in history[-1].content else BOOKED

    return agent
PY
